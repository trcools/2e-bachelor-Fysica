\subsection{Maschke's Theorem}

\begin{theorem}[Maschke's Theorem]
    Elke representatie van een eindige groep \(G\) is equivalent aan de directe som van irreps zonder triviale deelruimtes.
\end{theorem}
\begin{proof}
    Stel we hebben een representatie van een rechts-invariante deelruimte van de vorm:
    \[
    A(r) = \begin{pmatrix}
        B(r) & 0 \\
        D(r) & C(r)
    \end{pmatrix}
    \]
    met de eigenschap dat \(D(ab) = D(a)B(b) + C(a)D(b)\), nu willen we een gelijksvormigheidstransformatie vinden van de vorm:
    \[
    \begin{pmatrix}
        \mathds{1} & 0 \\
        -F & \mathds{1}
    \end{pmatrix}
    \begin{pmatrix}
        B(r) & 0 \\
        D(r) & C(r)
    \end{pmatrix}
    \begin{pmatrix}
        \mathds{1} & 0 \\
        F & \mathds{1}
    \end{pmatrix}
    =
    \begin{pmatrix}
        B(r) & 0 \\
        0 & C(r)
    \end{pmatrix}
    \]
    Dit is equivalent aan de volgende vergelijking:
    \[D(r) = F B(r) - C(r) F\]
    We kunnen \(F\) vinden door te sommeren over alle elementen van de groep:
    \[F = \frac{1}{|G|} \sum_{g\in G} D(g) B(g)^{-1}\]
    We zullen nu aantonen dat deze \(F\) voldoet aan de vergelijking \(D(r) = F B(r) - C(r) F\):

    \begin{align*}
        F B(r) - C(r) F &= \frac{1}{|G|} \sum_{g\in G} D(g) B(g)^{-1} B(r) - \frac{1}{|G|} \sum_{g\in G} C(r) D(g) B(g)^{-1} \\
        &= \frac{1}{|G|} \sum_{g\in G} D(g) B(g)^{-1} B(r) - \frac{1}{|G|} \sum_{g\in G} \left(D(rg)-D(r)B(g)\right) B(g)^{-1} \\
        &= \frac{1}{|G|} \sum_{g\in G} D(g) B(g)^{-1} B(r) - \frac{1}{|G|} \sum_{g\in G} D(rg) B(g)^{-1} + \frac{1}{|G|} \sum_{g\in G} D(r) B(g)B(g)^{-1} \\
        &= \frac{1}{|G|} \sum_{g\in G} D(g) B(g)^{-1} B(r) - \frac{1}{|G|} \sum_{h\in G} D(h) B(h)^{-1} B(r) + D(r) \\
        &= D(r)
    \end{align*}

    Dus \(F\) voldoet aan de vergelijking \(D(r) = F B(r) - C(r) F\), en we kunnen de representatie diagonalizeren zonder dat er triviale deelruimtes ontstaan.
\end{proof}
